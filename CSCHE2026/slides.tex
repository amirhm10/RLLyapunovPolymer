\begin{column}{0.43\textwidth}
\begin{block}{Why MPC is a strong baseline}
\begin{itemize}\raggedright
  \setlength\itemsep{0.45ex}
  \item Handles MIMO interactions and constraints.
  \item Solves a familiar receding-horizon optimization problem.
  \item Offset-free augmentation helps reject steady disturbances.

\begin{block}{Main limitation}
\begin{itemize}\raggedright
  \setlength\itemsep{0.45ex}
  \item The fixed linear model is local to the identified region.
  \item Away from nominal operation, mismatch can degrade tracking.
\end{itemize}
\end{block}

\scriptsize

\vspace{-0.22cm}

\begin{columns}[T,onlytextwidth]

\begin{column}{0.38\textwidth}
\begin{block}{Idea}
\begin{itemize}\raggedright
  \setlength\itemsep{0.34ex}
  \item Use OF-MPC as an expert controller in offline mode.
  \item Collect expert transition tuples.
  \item Initialize the actor by imitation before online RL.
\end{itemize}
\end{block}

\vspace{0.03cm}

\begin{block}{Behavior-cloning loss}
\vspace{-0.12cm}
\[
s_i=\big[\hat z_i,\;y_{{\rm sp},i},\;u_{i-1}\big]
\]
\vspace{-0.28cm}
\[
\tau_i=(s_i,u_i^{\rm MPC},r_i,s_{i+1}),\quad
\mathcal D_{\rm IL}=\{\tau_i\}_{i=1}^{N}
\]
\vspace{-0.27cm}
\[
e_i=\pi_{\theta}(s_i)-u_i^{\rm MPC}
\]
\vspace{-0.27cm}
\[
\mathcal L_{\rm BC}(\theta)=
\frac{1}{N}\sum_{i=1}^{N}\operatorname{Huber}_{\delta}(e_i)
\]
\vspace{-0.20cm}
\end{block}

\vspace{-0.03cm}
{\refnote{Hassanpour et al., 2024; Hamedi et al., 2026}}
\end{column}

\begin{frame}
\setbeamertemplate{blocks}[rounded][shadow=false]

\frametitle{\large \textbf{Safety Check: One-Step Lyapunov Contraction}}
\scriptsize

\vspace{-0.12cm}
\begin{columns}[T,onlytextwidth]

\begin{column}{0.39\textwidth}
\begin{block}{Gate calculation}
\vspace{-0.12cm}
\[
 e_k=\hat x_k-x_s,\qquad V_k=e_k^\top P_x e_k
\]
\vspace{-0.24cm}
\[
V_{k+1}^{\rm cand}\le \rho V_k+\epsilon_{\rm lyap},
\qquad 0<\rho<1
\]
\vspace{-0.16cm}
\end{block}

\vspace{0.04cm}
\begin{block}{Decision}
\begin{itemize}\raggedright
  \setlength\itemsep{0.36ex}
  \item \textbf{Pass:} apply the candidate RL action.
  \item \textbf{Fail:} replace it with fallback LMPC.
\end{itemize}
\end{block}

\vspace{0.04cm}
\begin{block}{Interpretation}
\justifying
The candidate is accepted only if the predicted next state moves toward the local target in the Lyapunov metric.
\end{block}

\vspace{0.03cm}
\end{column}

\begin{itemize}\raggedright

  \item \textbf{Fallback LMPC:} computes a tracking move that satisfies the contraction condition.

\vspace{-0.10cm}

\[
\text{s.t.}\quad
u_{\min}\le u\le u_{\max},
\]
\[
V_{1|k}\le \rho V_k+\epsilon_{\rm lyap}.

\vspace{-0.18cm}
\end{block}
